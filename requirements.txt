from flask import Flask, render_template, request, jsonify
import pickle, os

app = Flask(__name__)

# Load model if exists, else train dummy
if os.path.exists('models/model.pkl'):
    model = pickle.load(open('models/model.pkl','rb'))
else:
    model = None

@app.route('/')
def home():
    return render_template('index.html')

@app.route('/predict', methods=['POST'])
def predict():
    from sklearn.ensemble import RandomForestClassifier
    import numpy as np

    d = request.json
    # Simple logic if model not trained yet
    # Home advantage formula that works immediately
    home_score = d['home_goals']*0.6 + d['home_shots']*0.05 + d['home_sot']*0.1
    away_score = d['away_goals']*0.6 + d['away_shots']*0.05 + d['away_sot']*0.1
    diff = home_score - away_score

    if diff > 0.4:
        pred, conf, ph, pd, pa = "Home Win", 65, 65, 10, 25
    elif diff < -0.4:
        pred, conf, ph, pd, pa = "Away Win", 62, 25, 13, 62
    else:
        pred, conf, ph, pd, pa = "Draw", 55, 35, 30, 35

    # If real model exists, use it
    if model:
        try:
            feats = [[d['home_goals'], d['away_goals'], d['home_shots'], d['away_shots'], d['home_sot'], d['away_sot']]]
            p = model.predict(feats)[0]
            proba = model.predict_proba(feats)[0]
            mapping = {1:"Home Win", 0:"Draw", 2:"Away Win"}
            pred = mapping.get(p, pred)
            conf = round(max(proba)*100,1)
            ph = round(proba[1]*100,1) if len(proba)>1 else ph
            pd = round(proba[0]*100,1)
            pa = round(proba[2]*100,1) if len(proba)>2 else pa
        except: pass

    return jsonify({"prediction": pred, "confidence": conf, "probabilities": {"home": ph, "draw": pd, "away": pa}})

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000, debug=True)
